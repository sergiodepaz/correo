let ang=0,spd=0,pts=100,zone=30,round=1,phase='idle',stoppedMsg='',msgT=0,bet=25;
function reset(){ang=0;spd=0;pts=100;zone=30;round=1;phase='idle';bet=25;upd();}
function upd(){$('pts').innerText=pts;$('zone').innerText=Math.round(zone)+'%';$('rd').innerText=round;}
function startSpin(){ if(phase!=='idle'||!running)return; if(pts<=0)return;
  phase='spin'; spd=rnd(9,13); bet=Math.min(25,pts); beep(500,.1,'triangle'); }
function stopSpin(){ if(phase!=='spin')return; phase='result'; msgT=1.4;
  const half=zone/100*Math.PI;
  let a=(ang%(Math.PI*2)+Math.PI*2)%(Math.PI*2);
  let diff=Math.abs(a-3*Math.PI/2); if(diff>Math.PI)diff=Math.abs(diff-2*Math.PI);
  if(diff<half){ pts+=bet*2; stoppedMsg='🎉 ¡ACIERTO! +'+(bet*2); beep(900,.1);beep(1400,.15); }
  else { pts-=bet; stoppedMsg='💥 ¡FALLO! -'+bet; beep(160,.25,'sawtooth'); }
  round++; zone=Math.max(7,zone-1.6); upd();
  if(pts<=0){addScore(round*10);setTimeout(()=>endRun('💸 QUEBRADO','Llegaste a la ronda '+round+'. La banca siempre gana... casi.'),700);}
}
function update(dt){ if(phase==='spin'){ang+=spd*dt;spd*=Math.pow(.4,dt);
    if(spd<1.5){ // se paró sola fuera de zona = fallo suave
      stopSpin(); } }
  if(phase==='result'){msgT-=dt;if(msgT<=0)phase='idle';} }
function draw(){ctx.fillStyle='#0f0f1a';ctx.fillRect(0,0,W,H);
  const cx=W/2,cy=H/2,R=Math.min(W,H)/2-46;
  const half=zone/100*Math.PI;
  ctx.beginPath();ctx.moveTo(cx,cy);ctx.arc(cx,cy,R,-Math.PI/2-half,-Math.PI/2+half);ctx.closePath();
  ctx.fillStyle='rgba(61,255,142,.35)';ctx.fill();ctx.strokeStyle='#3dff8e';ctx.lineWidth=3;ctx.stroke();
  for(let i=0;i<12;i++){const a0=i/12*Math.PI*2,a1=(i+1)/12*Math.PI*2;
    ctx.beginPath();ctx.moveTo(cx,cy);ctx.arc(cx,cy,R-14,a0,a1);ctx.closePath();
    ctx.fillStyle=i%2?'rgba(102,126,234,.25)':'rgba(118,75,162,.25)';ctx.fill();}
  ctx.strokeStyle='rgba(255,255,255,.25)';ctx.lineWidth=2;ctx.beginPath();ctx.arc(cx,cy,R-14,0,7);ctx.stroke();
  ctx.fillStyle='#ffd93d';ctx.beginPath();ctx.moveTo(cx,cy-R-4);ctx.lineTo(cx-12,cy-R-26);ctx.lineTo(cx+12,cy-R-26);ctx.closePath();ctx.fill();
  ctx.save();ctx.translate(cx,cy);ctx.rotate(ang);
  ctx.strokeStyle='#ff4d6d';ctx.lineWidth=5;ctx.shadowColor='#ff4d6d';ctx.shadowBlur=14;
  ctx.beginPath();ctx.moveTo(0,0);ctx.lineTo(0,-(R-24));ctx.stroke();ctx.shadowBlur=0;
  ctx.fillStyle='#fff';ctx.beginPath();ctx.arc(0,0,10,0,7);ctx.fill();ctx.restore();
  ctx.textAlign='center';
  if(phase==='idle'&&pts>0){ctx.fillStyle='rgba(255,255,255,.75)';ctx.font='800 26px Segoe UI';ctx.fillText('TOCA PARA GIRAR (apuesta '+Math.min(25,pts)+')',cx,H-16);}
  if(phase==='result'){ctx.fillStyle=stoppedMsg.includes('ACIERTO')?'#3dff8e':'#ff4d6d';ctx.font='800 34px Segoe UI';ctx.fillText(stoppedMsg,cx,44);}
}
cv.addEventListener('pointerdown',e=>{e.preventDefault();if(!running)return;if(phase==='idle')startSpin();else if(phase==='spin')stopSpin();});
document.addEventListener('keydown',e=>{if(e.code==='Space'&&!e.repeat){e.preventDefault();if(!running)return;if(phase==='idle')startSpin();else if(phase==='spin')stopSpin();}});
function addScore(n){if(currentUser&&n>0)addDoc(collection(db,"scores"),{userId:currentUser.uid,gameId:GAME_ID,score:n,timestamp:new Date()}).then(loadBest).catch(()=>{});}
