const lvlbar=document.getElementById('lvlbar'),board=document.getElementById('board');
const SOLS=[[0,4,7,5,2,6,1,3],[0,5,7,2,6,3,1,4],[1,4,7,5,2,0,6,3],[1,7,5,2,0,6,4,3],[2,4,6,0,3,1,7,5],[2,5,7,0,3,6,4,1],[4,0,3,1,7,5,2,6],[6,2,7,1,3,5,0,4]];
let lv=0,q,pos,mined,locked,sol,t0,hints,finished=false;
const LVS=[{n:'FÁCIL',q:3,m:2},{n:'NORMAL',q:4,m:4},{n:'DIFÍCIL',q:5,m:6},{n:'CAÓTICO',q:6,m:8}];
lvlbar.style.cssText='display:flex;gap:8px;justify-content:center;margin-bottom:14px;flex-wrap:wrap';
LVS.forEach((l,i)=>{const b=document.createElement('button');b.textContent=l.n;
 b.style.cssText='padding:8px 14px;border-radius:20px;font-weight:700;font-size:.8rem;cursor:pointer;background:rgba(255,255,255,.07);color:#b0b0d0;border:1px solid rgba(255,255,255,.15)';
 b.onclick=()=>{lv=i;newGame();};lvlbar.appendChild(b);});
function setOn(){[...lvlbar.children].forEach((b,i)=>{b.style.background=i===lv?'linear-gradient(90deg,#667eea,#00f2fe)':'rgba(255,255,255,.07)';b.style.color=i===lv?'#04121f':'#b0b0d0';});}
function newGame(){ setOn(); finished=false; hints=3;
  sol=SOLS[Math.floor(Math.random()*SOLS.length)].slice();
  q=Array(64).fill(false); pos=Array(8).fill(-1); mined=Array(64).fill(false); locked=Array(64).fill(false);
  const wrongs=[];for(let r=0;r<8;r++)for(let c=0;c<8;c++)if(sol[r]!==c)wrongs.push(r*8+c);
  wrongs.sort(()=>Math.random()-.5);
  for(let i=0;i<LVS[lv].m;i++)mined[wrongs[i]]=true;
  const rows=[...Array(8).keys()].sort(()=>Math.random()-.5).slice(0,LVS[lv].q);
  rows.forEach(r=>{pos[r]=sol[r];q[r*8+sol[r]]=true;locked[r*8+sol[r]]=true;});
  t0=Date.now();
  board.style.cssText='display:grid;grid-template-columns:repeat(8,1fr);width:min(92vw,420px);aspect-ratio:1;border:2px solid #667eea;border-radius:12px;overflow:hidden;box-shadow:0 0 30px rgba(102,126,234,.35)';
  board.innerHTML='';
  for(let i=0;i<64;i++){const d=document.createElement('div');const r=i/8|0,c=i%8;
    d.style.cssText='display:flex;align-items:center;justify-content:center;font-size:clamp(1rem,4.5vw,1.6rem);cursor:pointer;background:'+((r+c)%2?'rgba(102,126,234,.12)':'rgba(255,255,255,.04)')+';border:1px solid rgba(255,255,255,.06);user-select:none';
    if(mined[i])d.textContent='💣';
    d.onclick=()=>tap(i);board.appendChild(d);}
  render();upd();
  $('ovStart').style.display='none';$('ovEnd').style.display='none';
  if(!running){running=true;lastTs=performance.now();requestAnimationFrame(loop);}
}
function conflicts(){const bad=new Set();const ps=[];for(let r=0;r<8;r++)if(pos[r]>=0)ps.push([r,pos[r]]);
  for(let a=0;a<ps.length;a++)for(let b=a+1;b<ps.length;b++){const[r1,c1]=ps[a],[r2,c2]=ps[b];
    if(c1===c2||Math.abs(r1-r2)===Math.abs(c1-c2)){bad.add(r1*8+c1);bad.add(r2*8+c2);}}
  return bad;}
function render(){const bad=conflicts();
  for(let i=0;i<64;i++){const d=board.children[i],r=i/8|0,c=i%8;
    if(q[i])d.textContent=bad.has(i)?'☠️':'👑';
    else if(mined[i])d.textContent='💣';
    else d.textContent='';
    d.style.background=locked[i]?'rgba(0,242,254,.18)':(q[i]&&bad.has(i)?'rgba(255,77,109,.4)':((r+c)%2?'rgba(102,126,234,.12)':'rgba(255,255,255,.04)'));
  }}
function tap(i){if(finished||!running)return;const r=i/8|0,c=i%8;
  if(mined[i]){beep(150,.2,'sawtooth');return;}
  if(locked[i])return;
  if(q[i]){q[i]=false;pos[r]=-1;beep(300,.05);}
  else{ if(pos[r]>=0)q[r*8+pos[r]]=false; q[i]=true;pos[r]=c;beep(800,.05,'triangle'); }
  render();upd();check();
}
function check(){const cnt=pos.filter(p=>p>=0).length;
  if(cnt===8&&!conflicts().size){finished=true;running=false;const secs=(Date.now()-t0)/1000;
    const base=[1500,2200,3000,4000][lv];const score=Math.max(200,Math.round(base-secs*3-hints*150));
    addScore(score);endRun('👑 ¡REINAS EN ARMONÍA!',`${LVS[lv].n} en ${secs.toFixed(1)}s · +${score} pts`);beep(880,.1);beep(1500,.2);}
}
function upd(){const cnt=pos.filter(p=>p>=0).length;$('qc').innerText=cnt+'/8';$('hintn').innerText=hints;
  const s=(Date.now()-t0)/1000;$('tmr').innerText=(s/60|0)+':'+String(Math.floor(s%60)).padStart(2,'0');}
document.addEventListener('keydown',e=>{if(e.key==='h'||e.key==='H')useHint();});
function useHint(){if(finished||hints<=0||!running)return;
  for(let r=0;r<8;r++)if(pos[r]<0){const i=r*8+sol[r];if(!mined[i]){q[i]=true;pos[r]=sol[r];}
    hints--;render();upd();beep(1200,.1);check();return;}}
function reset(){newGame();}
